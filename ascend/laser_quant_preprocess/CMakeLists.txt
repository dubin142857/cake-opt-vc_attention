cmake_minimum_required(VERSION 3.20)
project(vc_laser_quant_preprocess LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(RUN_MODE "npu" CACHE STRING "Ascend execution mode")
set(SOC_VERSION "${SOC_VERSION}" CACHE STRING "Exact A2/A3 compiler SoC")
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_CANN_PACKAGE_PATH}" CACHE PATH "CANN toolkit")
if(NOT SOC_VERSION MATCHES "^Ascend910B([1-4]|2C)$" AND NOT SOC_VERSION MATCHES "^Ascend910_93[0-9][0-9]$")
  message(FATAL_ERROR "Set exact A2/A3 SOC_VERSION from check_ascend910b.py")
endif()
foreach(candidate tools/tikcpp/ascendc_kernel_cmake compiler/tikcpp/ascendc_kernel_cmake ascendc_devkit/tikcpp/samples/cmake)
  if(EXISTS "${ASCEND_CANN_PACKAGE_PATH}/${candidate}/ascendc.cmake")
    set(ASCENDC_CMAKE_DIR "${ASCEND_CANN_PACKAGE_PATH}/${candidate}")
    break()
  endif()
endforeach()
if(NOT ASCENDC_CMAKE_DIR)
  message(FATAL_ERROR "CANN ascendc.cmake not found; source complete development toolkit")
endif()
include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import torch; print(torch.utils.cmake_prefix_path)"
  OUTPUT_VARIABLE torch_prefix OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
find_package(Torch REQUIRED HINTS "${torch_prefix}")
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import pathlib, torch_npu; print(pathlib.Path(torch_npu.__file__).parent)"
  OUTPUT_VARIABLE torch_npu_path OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
separate_arguments(torch_flags NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
ascendc_library(vc_quantize_kernel STATIC quantize_kernel.cpp)
ascendc_include_directories(vc_quantize_kernel PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${ASCEND_CANN_PACKAGE_PATH}/include")
add_library(vc_laser_quant_preprocess SHARED quantize_torch.cpp)
target_compile_options(vc_laser_quant_preprocess PRIVATE ${torch_flags})
target_compile_definitions(vc_laser_quant_preprocess PRIVATE VC_QUANT_SOC_VERSION="${SOC_VERSION}")
target_include_directories(vc_laser_quant_preprocess PRIVATE ${TORCH_INCLUDE_DIRS}
  "${torch_npu_path}/include" "${ASCEND_CANN_PACKAGE_PATH}/include")
target_link_directories(vc_laser_quant_preprocess PRIVATE "${torch_npu_path}/lib" "${ASCEND_CANN_PACKAGE_PATH}/lib64")
target_link_libraries(vc_laser_quant_preprocess PRIVATE vc_quantize_kernel ${TORCH_LIBRARIES}
  torch_npu ascendcl tiling_api platform register graph_base dl m)
set_target_properties(vc_laser_quant_preprocess PROPERTIES BUILD_RPATH "${torch_npu_path}/lib;${ASCEND_CANN_PACKAGE_PATH}/lib64")
