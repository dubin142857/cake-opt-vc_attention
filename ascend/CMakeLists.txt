cmake_minimum_required(VERSION 3.20)
project(vc_attention_ascend LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(RUN_MODE "npu" CACHE STRING "Ascend execution mode")
set(SOC_VERSION "${SOC_VERSION}" CACHE STRING "Exact compiler SoC configuration")
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_CANN_PACKAGE_PATH}" CACHE PATH "CANN toolkit root")
if(NOT SOC_VERSION MATCHES "^Ascend910B([1-4]|2C)$" AND NOT SOC_VERSION MATCHES "^Ascend910_93[0-9][0-9]$")
  message(FATAL_ERROR "Set SOC_VERSION to the exact A2/A3 DAV-2201 SoC returned by preflight; do not use npu-smi marketing names")
endif()
foreach(candidate tools/tikcpp/ascendc_kernel_cmake compiler/tikcpp/ascendc_kernel_cmake ascendc_devkit/tikcpp/samples/cmake)
  if(EXISTS "${ASCEND_CANN_PACKAGE_PATH}/${candidate}/ascendc.cmake")
    set(ASCENDC_CMAKE_DIR "${ASCEND_CANN_PACKAGE_PATH}/${candidate}")
    break()
  endif()
endforeach()
if(NOT ASCENDC_CMAKE_DIR)
  message(FATAL_ERROR "CANN ascendc.cmake not found. Install/source the complete Ascend C development toolkit.")
endif()
include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import torch; print(torch.utils.cmake_prefix_path)"
  OUTPUT_VARIABLE torch_prefix OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
find_package(Torch REQUIRED HINTS "${torch_prefix}")
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import pathlib, torch_npu; print(pathlib.Path(torch_npu.__file__).parent)"
  OUTPUT_VARIABLE torch_npu_path OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
separate_arguments(torch_flags NATIVE_COMMAND "${TORCH_CXX_FLAGS}")

ascendc_library(vc910_kernel STATIC csrc/vc_kernel.cpp)
ascendc_include_directories(vc910_kernel PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/csrc" "${ASCEND_CANN_PACKAGE_PATH}/include")
add_library(vc_ascend_ops SHARED csrc/vc_torch.cpp)
target_compile_options(vc_ascend_ops PRIVATE ${torch_flags})
target_compile_definitions(vc_ascend_ops PRIVATE VC_ASCEND_SOC_VERSION="${SOC_VERSION}")
target_include_directories(vc_ascend_ops PRIVATE csrc ${TORCH_INCLUDE_DIRS}
  "${torch_npu_path}/include" "${ASCEND_CANN_PACKAGE_PATH}/include")
target_link_directories(vc_ascend_ops PRIVATE "${torch_npu_path}/lib" "${ASCEND_CANN_PACKAGE_PATH}/lib64")
target_link_libraries(vc_ascend_ops PRIVATE vc910_kernel ${TORCH_LIBRARIES} torch_npu
  ascendcl tiling_api platform register graph_base dl m)
set_target_properties(vc_ascend_ops PROPERTIES BUILD_RPATH "${torch_npu_path}/lib;${ASCEND_CANN_PACKAGE_PATH}/lib64")
